# Probe programs (SPEC §9 rule 5): small programs that observe the real behaviour of MLT, Qt and the desktop.
option(VELACUT_BUILD_PROBES "Build the probe programs in tools/probes" OFF)
if(VELACUT_BUILD_PROBES)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(MLTPP REQUIRED IMPORTED_TARGET mlt++-7)
    add_executable(mlt_probe probes/mlt_probe.cpp)
    target_link_libraries(mlt_probe PRIVATE PkgConfig::MLTPP)
    add_executable(projection_probe probes/projection_probe.cpp)
    target_link_libraries(projection_probe PRIVATE PkgConfig::MLTPP)
    add_executable(phase2_probe probes/phase2_probe.cpp)
    target_link_libraries(phase2_probe PRIVATE PkgConfig::MLTPP Qt6::Gui)
    add_executable(preview_probe probes/preview_probe.cpp)
    target_link_libraries(preview_probe PRIVATE PkgConfig::MLTPP)
    add_executable(appearance_probe probes/appearance_probe.cpp)
    target_link_libraries(appearance_probe PRIVATE velacut_theme)
    velacut_target_options(appearance_probe)
endif()
